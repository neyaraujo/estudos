	Operado		Significado		Exemplo		Resultado
	==		igual(valor)		5 == "5"	true
	===		idêntico(valor + tipo)	5 === "5"	false
	!=		diferente		5 != 3		true
	<>		diferente		5 <> 3		true
	!==		não idêntico		5 !== "5"	true
	>		maior que		10 > 5		true
	<		menor que		3 < 7		true
	>=		maior ou igual		5 >= 5		true
	<=		maenor ou igual		2 <= 4		true

|---	Exemplo
		5 == "5" 	// saida: true
		5 === "5" 	// saida: false

|---	Operador null coalescing (??) - verfica se EXISTE e se NÃO é null
		valores que NAO ativam o ??
			'' | 0 | '0' | false | []
		valor que ativam o ??
			inexistente, null

		Substitui valores quando não existem ou são NULL
		Exemplo
			Sintaxe
				$valor = $variavel ?? 'padrao'; // Se $variavel existir e não for null, use ela, senão use 'padrão'.
			exemplo
				$nome = $_POST['nome'] ?? 'visitante';
			equivalente a
				if(isset($_POST['nome'])) {
					$nome = $_POST['nome'];
				} else {
					$nome = 'Visitante';
				}
				
			exemplo
				$nome = $_POST['nome'] ?? "teste";
			equivalente a:
				$nome = isset($_POST['nome']) ? $_POST['nome'] : "teste";

|---	Operador Ternário Completo
		Sintaxe	
			condição ? valor_se_verdadeiro : valor_se_falso;
		exemplo
			$idade = 20
			$resultado = $idade >= 18 ? 'Maior' ? 'Menor';
			echo $resultado

|---	Ternário Simplificado (?:) - verifica se o valor é verdadeiro ou falso
		Valores falsos
			false | 0 | 0.0 | '' | '0' | null | []
		
		Sintaxe
			$valor = $variavel ?: 'padrão';
			$nome = '' ?: 'Visitante'; // saida: Visitante;

|---	Duvidas entre (??) & (?:)
		com: (?? :)
			$nome = $_POST['nome'] ?? 'Visitante'; // Se vier ""(vazio), resultado é ""(vazio ) por que existe.
		com: (?:)
			$nome = $_POST['nome'} ?: 'Visitante'; // Se vier ""(vazio), resultado é Visitante, por string vazia é falsa.

#SESSAO
	O que é
		Mecanismo para armazenar dados do usuário entre páginas
		Permite manter informações enquanto o usuário navega no sistema

	Função principal
		session_start()

	Sintaxe
		session_start();
		$_SESSION['chave'] = valor;

	Exemplo
		session_start();

		$_SESSION['usuario'] = "Ney";
		$_SESSION['nivel'] = "admin";

		echo $_SESSION['usuario'];

	Observações
		Precisa iniciar a sessão antes de usar $_SESSION
		Geralmente usado em login, carrinho e controle de usuário
		A sessão funciona em todo o projeto enquanto estiver ativa
		Os dados ficam armazenados no servidor

	Funções importantes
		session_start()
			Inicia ou recupera uma sessão

		session_destroy()
			Destrói a sessão inteira

		unset($_SESSION['chave'])
			Remove um item específico da sessão

		session_unset()
			Limpa todas as variáveis da sessão

	Erros comuns
		Esquecer o session_start()
		Enviar HTML antes do session_start()
		Tentar acessar $_SESSION sem iniciar sessão
		Confundir sessão com cookie

	Exemplo de erro
		echo "Olá";
		session_start();

	Problema
		session_start() deve vir antes de qualquer saida HTML

	Exemplo correto
		session_start();
		echo "Olá";
