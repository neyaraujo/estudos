|---	ParentElement
	palavra chave
		Pega o pai do elemento, ancestrais
		</> html
			<div> classe"card">
				<button id="btn"><clique</button>
			</div>
		</> javascript
			const btn = document.getElementById('btn');
			cont pai =  btn.parentElement;
			pai.style.backgroundColor = 'lightblue';
			//resultado: o <div class="card"> recebe o estilo

	ParentNode
		Parecido com parentelement, mas pode retornar nós que não são elementos HTML
		const pai = btn.parentNode;
	closest() forma moderna e recomendada
		sobe na árvore procurando um ancestral específico.
		exemplo
			</>  html
				<div class"container">
					<div class"card">
						<button class"btn">Salvar</button>
					</div>
				</div>
			</> javascript
				const botao = document.querySelector('.btn');
				cont card = botao.closet('.card');

|--	Exemplo moderno com closest()
		document.addEventListener('click', (e) => {
			const btn = e.target.closest('.btn');
			if (!btn) return;
			const card = btn.closest('.card');
			card.classList.toggle('ativo');
		});
