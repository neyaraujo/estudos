#SESSAO
	O que é
		Mecanismo para armazenar dados do usuário entre páginas
		Permite manter informações enquanto o usuário navega no sistema

	Função principal
		session_start()

	Sintaxe
		session_start();
		$_SESSION['chave'] = valor;

	Exemplo
		session_start();

		$_SESSION['usuario'] = "Ney";
		$_SESSION['nivel'] = "admin";

		echo $_SESSION['usuario'];

	Observações
		Precisa iniciar a sessão antes de usar $_SESSION
		Geralmente usado em login, carrinho e controle de usuário
		A sessão funciona em todo o projeto enquanto estiver ativa
		Os dados ficam armazenados no servidor

	Funções importantes
		session_start()
			Inicia ou recupera uma sessão

		session_destroy()
			Destrói a sessão inteira

		unset($_SESSION['chave'])
			Remove um item específico da sessão

		session_unset()
			Limpa todas as variáveis da sessão

	Erros comuns
		Esquecer o session_start()
		Enviar HTML antes do session_start()
		Tentar acessar $_SESSION sem iniciar sessão
		Confundir sessão com cookie

	Exemplo de erro
		echo "Olá";
		session_start();

	Problema
		session_start() deve vir antes de qualquer saida HTML

	Exemplo correto
		session_start();
		echo "Olá";
